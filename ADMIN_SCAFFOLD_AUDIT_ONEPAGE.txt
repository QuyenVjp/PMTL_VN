QUERY/MUTATION PATTERN (For All Features)
═══════════════════════════════════════════════════════════════════════════════

// query-keys.ts — Factory functions (REQUIRED)
export const {feature}Keys = {
  all: ['{feature}'] as const,
  lists: () => [{feature}, 'list'] as const,
  list: (filters) => [{feature}, 'list', filters] as const,
  details: () => [{feature}, 'detail'] as const,
  detail: (id) => [{feature}, 'detail', id] as const,
};

// queries.ts — Read operations
export const use{Feature}List = (filters?) => useQuery({
  queryKey: {feature}Keys.list(filters),
  queryFn: () => api.get('/api/...'),
});

// mutations.ts — Write operations + invalidation
export const useCreate{Feature} = () => {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (data) => api.post('/api/...', data),
    onSuccess: (data) => {
      // Per canonical rules from admin-page-api-mapping.md
      queryClient.invalidateQueries({ queryKey: {feature}Keys.lists() });
      queryClient.invalidateQueries({ queryKey: {feature}Keys.detail(data.id) });
      // Dashboard cascade if applicable
      queryClient.invalidateQueries({ queryKey: ['admin-dashboard', 'recent-posts'] });
      // Public cache trigger if publish
      if (data.isPublished) {
        await api.post('/internal/revalidate', { source: 'feature-publish', id: data.id });
      }
    },
  });
};
